## PART 2 (blocking gate) -- full code

```bash
check_spec_currency_gate() {
    local _cg_fail=0
    local _cg_staged_files
    _cg_staged_files=$(git diff --cached --name-only --diff-filter=ACMR 2>/dev/null | grep -E '\.(ps1|cs|py|sh|ts|js|go|rs|java|rb|php|sql)$')
    [ -z "$_cg_staged_files" ] && return 0
    local _cg_repo_root; _cg_repo_root=$(git rev-parse --show-toplevel 2>/dev/null) || return 0
    local _cg_claude_path="$_cg_repo_root/CLAUDE.md"
    [ ! -f "$_cg_claude_path" ] && return 0
    local _cg_jsonl_path="$WKHARNESS_CLAUDE_PATH"
    if [ -z "$_cg_jsonl_path" ] && [ -d "$HOME/.claude/projects" ]; then
        _cg_jsonl_path=$(find "$HOME/.claude/projects" -name '*.jsonl' -type f -printf '%T@ %p\n' 2>/dev/null | sort -rn | head -1 | cut -d' ' -f2-)
    fi
    while IFS= read -r _cg_file; do
        [ -z "$_cg_file" ] && continue
        local _cg_declared_skills; _cg_declared_skills=$(grep -h '^\s*#///' "$_cg_repo_root/$_cg_file" 2>/dev/null | grep -oE 'wkappbot skill read [A-Za-z0-9_.-]+' | sed 's/.*skill read //' | sort -u)
        local _cg_rule_skills; _cg_rule_skills=$(awk -v pattern="$_cg_file" '/^## harness:skill / { rule_pattern=$3; in_block=1; next } in_block && /^## / { in_block=0; next } in_block && /wkappbot skill read ([A-Za-z0-9_.-]+)/ { match($0,/wkappbot skill read ([A-Za-z0-9_.-]+)/,m); print m[1] }' "$_cg_claude_path" | sort -u)
        local _cg_all_bound_skills; _cg_all_bound_skills=$(printf "%s\n%s\n" "$_cg_declared_skills" "$_cg_rule_skills" | grep -v '^$' | sort -u)
        [ -z "$_cg_all_bound_skills" ] && continue
        while IFS= read -r _cg_sid; do
            [ -z "$_cg_sid" ] && continue
            local _cg_was_edited=0
            if [ -n "$_cg_jsonl_path" ] && [ -f "$_cg_jsonl_path" ]; then
                grep -q "wkappbot skill edit.*$_cg_sid" "$_cg_jsonl_path" 2>/dev/null && _cg_was_edited=1
            fi
            if [ $_cg_was_edited -eq 0 ]; then
                echo "[pre-commit] BLOCK: bound skill $_cg_sid (for $_cg_file) was NOT edited this session." >&2
                _cg_fail=1; continue
            fi
            local _cg_score=0
            if command -v pwsh >/dev/null 2>&1; then
                _cg_score=$(pwsh -NoProfile -Command ". (Join-Path '$_cg_repo_root/tools' 'wkharness-common.ps1') 2>/dev/null; $r=Get-WkTierScore '$_cg_sid'; if($r -and $r.Score){[int]$r.Score}else{100}" 2>/dev/null)
            fi
            { [ -z "$_cg_score" ] || [ $_cg_score -lt 0 ]; } && _cg_score=100
            if [ $_cg_score -lt 60 ]; then
                echo "[pre-commit] BLOCK: skill $_cg_sid tier-score $_cg_score < 60." >&2; _cg_fail=1
            fi
        done <<< "$_cg_all_bound_skills"
    done <<< "$_cg_staged_files"
    [ $_cg_fail -ne 0 ] && { echo "[pre-commit] Commit blocked: spec-currency gate." >&2; return 1; }
    return 0
}
```
