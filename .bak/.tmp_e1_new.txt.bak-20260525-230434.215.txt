
    /// <summary>
    /// Process-only scan (NO HTTP probe) for a live chrome.exe whose --user-data-dir
    /// cmdline argument EQUALS our userDataDir. Returns (pid, port) parsed from the
    /// cmdline, or (0, 0) when none found. Last-resort recovery for when Chrome is
    /// process-alive but its CDP HTTP endpoint is momentarily unreachable so the
    /// HTTP-based probes all return 0. CmdLineUserDataDirEquals keeps per-CWD isolation.
    /// </summary>
    public static (int pid, int port) FindChromeProcessByUserDataDir(string userDataDir)
    {
        try
        {
            if (string.IsNullOrWhiteSpace(userDataDir)) return (0, 0);
            var normalizedDir = Path.GetFullPath(userDataDir).TrimEnd(BACKSLASH_FWDSLASH).ToLowerInvariant();
            foreach (var proc in SafeGetChromeProcesses())
            {
                try
                {
                    var cmd = GetCommandLine(proc.Id);
                    if (cmd == null) continue;
                    if (!CmdLineUserDataDirEquals(cmd, normalizedDir)) continue;
                    var m = System.Text.RegularExpressions.Regex.Match(
                        cmd, PORT_REGEX);
                    if (m.Success && int.TryParse(m.Groups[1].Value, out var p) && p > 0)
                        return (proc.Id, p);
                }
                catch { }
                finally { proc.Dispose(); }
            }
        }
        catch { }
        return (0, 0);
    }
