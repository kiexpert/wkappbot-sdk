                        // -- Tab-reuse probe (2026-05-20+) ------------------------------------
                        // List existing tabs and look for one already on the same URL/domain.
                        // If found, switch to it focuslessly (SwitchToTargetAsync minimizes
                        // Chrome before disconnect+reconnect, no SetForegroundWindow). Skips
                        // Target.createTarget entirely. Prevents tab accumulation when the
                        // same site is opened repeatedly. --new-tab opts out.
                        if (!forceNewTab)
                        {
                            try
                            {
                                string? matchHost = null;
                                try { matchHost = new Uri(url).Host; } catch { matchHost = null; }
                                if (!string.IsNullOrEmpty(matchHost))
                                {
                                    var tabs = probeCdp.ListTabsAsync(port).GetAwaiter().GetResult();
                                    WKAppBot.WebBot.CdpClient.TabInfo? hit = null;
                                    foreach (var t in tabs)
                                    {
                                        if (string.IsNullOrEmpty(t.Url)) continue;
                                        if (t.Url.Contains("#wkbot-", StringComparison.Ordinal)) continue;
                                        if (string.Equals(t.Url, url, StringComparison.OrdinalIgnoreCase) ||
                                            t.Url.StartsWith(url.TrimEnd('/'), StringComparison.OrdinalIgnoreCase))
                                        { hit = t; break; }
                                    }
                                    if (hit == null)
                                    {
                                        foreach (var t in tabs)
                                        {
                                            if (string.IsNullOrEmpty(t.Url)) continue;
                                            if (t.Url.Contains("#wkbot-", StringComparison.Ordinal)) continue;
                                            if (!Uri.TryCreate(t.Url, UriKind.Absolute, out var tabUri)) continue;
                                            if (string.Equals(tabUri.Host, matchHost, StringComparison.OrdinalIgnoreCase))
                                            { hit = t; break; }
                                        }
                                    }
                                    if (hit != null)
                                    {
                                        probeCdp.Dispose();
                                        Console.ForegroundColor = ConsoleColor.Green;
                                        Console.WriteLine($"[WEB] Reusing existing tab {hit.Id[..Math.Min(8,hit.Id.Length)]} for {url} (host={matchHost})");
                                        Console.ResetColor();

                                        using var cdpR = new WKAppBot.WebBot.CdpClient();
                                        cdpR.ConnectAsync(port, preferredTargetTag: hit.Id).GetAwaiter().GetResult();
                                        try { cdpR.SwitchToTargetAsync(hit.Id, port).GetAwaiter().GetResult(); } catch { }

                                        var titleR = cdpR.GetTitleAsync().GetAwaiter().GetResult() ?? "";
                                        var pageUrlR = cdpR.GetUrlAsync().GetAwaiter().GetResult() ?? "";
                                        var chromeHwndR = cdpR.GetChromeWindowHandle();
                                        cdpR.ChromeWindowHandle = (nint)chromeHwndR;
                                        Console.Error.WriteLine($"[WEB] Title: {titleR}");
                                        Console.Error.WriteLine($"[WEB] URL:   {pageUrlR}");
                                        Console.Error.WriteLine($"[WEB] TabID: {hit.Id}");
                                        PrintWebTarget(cdpR, port);
                                        if (!args.Contains("--no-minimize") && !args.Contains("--show"))
                                        {
                                            cdpR.MinimizeChromeWindow();
                                            Console.WriteLine("[WEB] Chrome minimized (CDP works in background)");
                                        }
                                        else
                                        {
                                            Console.WriteLine("[WEB] Chrome window visible");
                                        }
                                        if (!string.IsNullOrEmpty(pageUrlR))
                                            ShowDomainKnowhow(pageUrlR);
                                        LaunchAppBotEyeIfNeeded(port);
                                        return 0;
                                    }
                                }
                            }
                            catch (Exception ex)
                            {
                                Console.Error.WriteLine($"[WEB] tab-reuse probe non-fatal: {ex.Message}");
                            }
                        }

                        // No reusable tab found (or --new-tab) -- create a fresh one.
